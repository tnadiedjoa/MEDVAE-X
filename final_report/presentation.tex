\begin{frame}{Introduction}
  \begin{columns}[T]
  \begin{column}{0.58\textwidth}
  \textbf{Contexte.} L'apprentissage profond en imagerie médicale est limité
  par la haute résolution nécessaire au détail diagnostique. Les VAE
  compressent l'image en un latent compact.
  \begin{itemize}
    \item \textbf{MedVAE} : compresseur médical générique ($>1$M images),
      $\times 16$ ($512^2 \to 128^2 \times 1$), entraîné en deux étapes
      (reconstruction puis enrichissement sémantique).
  \end{itemize}
  \vspace{0.4em}
  \textbf{Domaine.} Angiographie coronaire (ARCADE, $1\,200$ images) :
  structures fines, tubulaires, peu contrastées, bruit et artefacts.
  \end{column}
  \begin{column}{0.42\textwidth}
  \begin{block}{Question centrale}
  Un compresseur médical générique reste-t-il \alert{fiable et utile} face à
  un domaine clinique spécifique et exigeant ?
  \end{block}
  \vspace{0.2em}
  \footnotesize \textbf{Quatre axes complémentaires :}
  \begin{enumerate}
    \item Robustesse aux dégradations 
    \item Biais inductif de qualité via FiLM 
    \item Structuration sémantique du latent (JEPA)
    \item Représentation latente pour la segmentation 
  \end{enumerate}
  \end{column}
  \end{columns}
\end{frame}

\section{Biais inductif de qualité d'image (FiLM)}

% Motivation
\begin{frame}{Motivation}
  \begin{itemize}
    \item MedVAE applies the \textbf{same transform} to a clean or a heavily
      degraded coronary angiography: no notion of \emph{how degraded} the
      input is.
    \item \textbf{Hypothesis:} conditioning MedVAE on a quality score
      $c \in [0,1]$, injected via FiLM, should let it adapt to the
      degradation level actually present.
    \item Three independent ways to compute $c$ (A: weighted, B: PCA,
      C: learned) $\rightarrow$ run the \emph{same} pipeline three times.
  \end{itemize}
\end{frame}

  \vfill
  {\small
  \[\boxed{
    \mathcal{L}_{\text{stage2}}
    = \mathcal{L}_{\text{pred}}
    + \lambda_{\text{SIGReg}}\,\mathcal{L}_{\text{SIGReg}}
  }\]
  \[
    \mathcal{L}_{\text{pred}}
    = \frac{1}{|\mathcal{M}|} \sum_{p \in \mathcal{M}}
      \mathrm{SmoothL_1}\!\left(\hat{z}_t^{(p)},\, \tilde{z}_t^{(p)}\right),
    \qquad
    \mathcal{L}_{\text{SIGReg}}
    = \mathbb{E}_{\mathbf{v}}\!\left[ T_{\text{EP}}\!\left(\bigl\{ z_c^\top \mathbf{v} \bigr\}\right) \right]
  \]}
\end{frame}

  \centering
  \includegraphics[width=\linewidth]{yanic/recon.png}

  \begin{itemize}
    \item Décodeur \alert{gelé} : tout déplacement de l'encodeur dégrade le PSNR (16.85 dB pour le stage 2).

\section{Intégration pour la segmentation multi-classes}

\begin{frame}{Contexte et problématique}
\begin{columns}[T]
\begin{column}{0.56\textwidth}
\textbf{Tâche.} Segmentation fine des artères coronaires sur ARCADE :
\begin{itemize}
  \item 26 classes (le fond et 25 segments)
  \item structures minces, tubulaires, peu contrastées
  \item déséquilibre extrême ($\approx 95\,\%$ de fond)
\end{itemize}
\vspace{0.6em}
\textbf{Outil.} MedVAE, auto-encodeur variationnel médical pré-entraîné : compression $\times 16$ ($512^2 \to 128^2 \times 1$), mais \alert{avec perte}.
\end{column}
\begin{column}{0.44\textwidth}
\begin{block}{Deux questions}
\begin{enumerate}
  \item La compression préserve-t-elle assez d'information anatomique pour la segmentation ?
  \item Où l'intégrer : espace \textbf{latent} ou espace \textbf{pixel} ?
\end{enumerate}
\end{block}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Méthode : cinq conditions}
\begin{columns}[T]
\begin{column}{0.52\textwidth}
\small
\begin{itemize}
  \item \textbf{A} — U-Net pleine résolution (référence)
  \item \textbf{A*} — U-Net de A sur images reconstruites, \alert{sans réentraînement} (contrôle)
  \item \textbf{B} — tête légère sur le latent gelé
  \item \textbf{C} — comme B, encodeur fine-tuné sur ARCADE
  \item \textbf{D} — U-Net réentraîné sur images reconstruites
\end{itemize}
\vspace{0.6em}
\footnotesize Espace \textbf{latent} : B, C \quad ; \quad espace \textbf{pixel} : A*, D.
\end{column}
\begin{column}{0.48\textwidth}
\centering
\includegraphics[width=\linewidth]{theo/cond_B.pdf}\\[0.5em]
\includegraphics[width=\linewidth]{theo/cond_D.pdf}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Résultat 1 : la segmentation depuis le latent échoue}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{itemize}
  \item B et C : Dice $\approx 0{,}04$ (contre 0,433 pour A)
  \item seul le fond est prédit, IoU $\to 0$
\end{itemize}
\textbf{Pourquoi ?}
\begin{itemize}
  \item déséquilibre $95\,\%$ de fond $\Rightarrow$ minimum trivial très attracteur
  \item latent à 1 canal, \alert{aucune caractéristique multi-échelles} (ni connexions de saut)
\end{itemize}
Le fine-tuning du compresseur (C) n'y change rien.
\end{column}
\begin{column}{0.52\textwidth}
\centering
\includegraphics[width=\linewidth]{theo/comparaison_conditions.png}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Perspectives}
  \begin{block}{Vers un \emph{foundation model} pour l'imagerie médicale par rayonnement}
  Unifier les axes explorés en un objectif d'entraînement unique.
  \end{block}
  \begin{itemize}
    \item \textbf{Objectif hybride JEPA + reconstruction} : prédire les patchs
      latents masqués tout en gardant l'encodeur ancré à un espace décodable.
    \item \textbf{Dégradations physiquement fondées} : bruit quantique,
      contraste/luminosité, champs multiplicatifs, flou de mouvement
      anisotrope (cardiaque/respiratoire).
    \item \textbf{Conditionnement (CVAE)} sur les métadonnées DICOM
      (modalité, dose, SNR, qualité) $\rightarrow$ représentations
      désenchevêtrées contenu / qualité d'acquisition.
    \item \textbf{Validation downstream} : classification, détection,
      segmentation multi-modalités (ARCADE comme banc d'essai exigeant).
  \end{itemize}
\end{frame}

\begin{frame}{Conclusion générale}
  \footnotesize
  Sur quatre axes complémentaires, nous avons sondé le comportement de MedVAE,
  compresseur médical générique, sur le domaine exigeant de l'angiographie coronaire :
  \begin{itemize}
    \item \textbf{Robustesse} : filtre passe-bas intelligent et débruiteur non
      linéaire ; s'effondre sur les hautes fréquences \emph{structurées}
      (artefacts JPEG) — la nature de la dégradation prime sur son intensité.
    \item \textbf{FiLM} : conditionner sur un score de qualité dégrade
      légèrement la reconstruction (compromis architectural), bien que le
      réseau exploite réellement le signal.
    \item \textbf{JEPA} : latent nettement plus compact ($90\%$ de variance sur
      $25$ dimensions), gain discriminatif, mais coût en reconstruction
      (décodeur gelé).
    \item \textbf{Segmentation} : la compression $\times 16$ préserve
      l'anatomie \alert{en espace pixel} (gain ciblé sur les vaisseaux fins) ;
      le latent à un canal reste inadapté à la prédiction dense directe.
  \end{itemize}
  \vspace{0.3em}
  \textbf{Bilan.} Compression fidèle réutilisable en espace pixel, mais latent
  pas encore assez \emph{organisé sémantiquement} ni \emph{conscient de
  l'acquisition} pour être segmenté ou conditionné directement.
\end{frame}
